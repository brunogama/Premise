```swift
import ConjectureCore

/// Observable engine lifecycle events for telemetry consumers.
/// These events are purely observational and MUST NOT affect
/// replay semantics or trace content.
public enum EngineEvent: Sendable {
  case runStarted(propertyID: PropertyIdentity, runIndex: Int)
  case replayAttempt(propertyID: PropertyIdentity, traceEntryCount: Int)
  case shrinkStep(propertyID: PropertyIdentity, iteration: Int)
  case runFinished(propertyID: PropertyIdentity, passed: Bool, runCount: Int)
}
```

Create `Sources/ConjectureTelemetry/TelemetrySink.swift`:
```swift
/// Protocol for observing engine events. Implementations MUST be
/// side-effect-free with respect to engine state (no mutation of
/// traces, configs, or results).
public protocol TelemetrySink: Sendable {
  func record(_ event: EngineEvent) async
}
```
